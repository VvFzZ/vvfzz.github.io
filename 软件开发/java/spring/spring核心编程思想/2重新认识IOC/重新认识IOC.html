PropertyDescriptor

import java.beans.*;

public class BeanIntegration {
    public static class User {
        private LocalDate birthDate;
        // getter/setter...
    }

    public static void main(String[] args) throws Exception {
        // 获取 BeanInfo
        BeanInfo beanInfo = Introspector.getBeanInfo(User.class);
        
        // 找到目标属性描述符
        for (PropertyDescriptor pd : beanInfo.getPropertyDescriptors()) {
            if ("birthDate".equals(pd.getName())) {
                // 设置自定义编辑器
                pd.setPropertyEditorClass(LocalDateEditor.class);
                
                // 使用编辑器
                PropertyEditor editor = pd.createPropertyEditor(null);
                editor.setAsText("1990-01-01");
                System.out.println("Birth date: " + editor.getValue());
            }
        }
    }
}

import java.beans.PropertyEditorSupport;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

// 自定义 LocalDate 编辑器
public class LocalDateEditor extends PropertyEditorSupport {
    private final DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd");
    
    @Override
    public void setAsText(String text) throws IllegalArgumentException {
        setValue(LocalDate.parse(text, formatter));
    }
    
    @Override
    public String getAsText() {
        LocalDate date = (LocalDate) getValue();
        return date != null ? date.format(formatter) : "";
    }
}

依赖查找和依赖注入对比

依赖查找主动获取，实现相对复杂，代码入侵性强，依赖容器API，可读性良好
依赖注入被动提供，实现便利，地入侵性，不依赖容器API，可读性一般

没有绝对好坏，只有相对合理；互补使用。

构造器注入 vs setter注入

必须的依赖使用构造器
可选依赖使用setter
