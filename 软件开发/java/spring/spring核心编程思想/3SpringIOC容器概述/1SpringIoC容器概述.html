BeanFactory和ApplicationContext谁才是真正的容器

BeanFactory是底层IOC容器，ApplicationContext组合了一个BeanFactory对象，在它的基础上增加了新特性（事件支持，AOP，国际化）
虽然ApplicationContext本身实现BeanFactory接口但它实际上使用的容器是通过组合来的（类似代理模式）；
详情如下：
AbstractRefreshableApplicationContext 组合了一个DefaultListableBeanFactory对象
ApplicationContext的beanFactory并不是他本身，而是组合的DefaultListableBeanFactory
继承关系如下：
ConfigurableApplicationContext接口中定义getBeanFactory方法
AbstractApplicationContext继承ConfigurableApplicationContext 定义抽象方法getBeanFactory
AbstractRefreshableApplicationContext继承AbstractApplicationContext，实现getBeanFactory，返回组合的DefaultListableBeanFactory

public class IoCContainerDemo{

    public static void main(String[] args){
        // 创建 BeanFactory 容器
        DefaultListableBeanFactory beanFactory = new DefaultListableBeanFactory();
        XmlBeanDefinitionReader reader = new XmlBeanDefinitionReader(beanFactory);
        //XML 配置文件 ClassPath 路径
        String location ="classpath:/META-INF/dependency-lookup-context.xml";
        // 加载配置
        int beanDefinitionsCount =reader.loadBeanDefinitions(location);
        System.out.println(“Bean 定义加载的数量:"+ beanDefinitionsCount);
        lookupCollectionByType(beanFactory);
    }

    private static void lookupCollectionByType(BeanFactory beanFactory){
        if(beanFactory instanceof ListableBeanFactory){
            ListableBeanFactory listableBeanfactory =(ListableBeanFactory) beanfactory;
            Map<String,User> users = listableBeanFactory.getBeansOfType(User.class);
            System.out.println("查找到的所有的 User 集合对象:"+ users);
        }
    }
}

FactoryBean

基础用法

public class CarFactoryBean implements FactoryBean<Car> {
    private String carInfo;  // 可配置属性
    
    @Override
    public Car getObject() throws Exception {
        Car car = new Car();
        String[] infos = carInfo.split(",");
        car.setBrand(infos[0]);
        car.setMaxSpeed(Integer.parseInt(infos[1]));
        car.setPrice(Double.parseDouble(infos[2]));
        return car;
    }
    
    @Override
    public Class<?> getObjectType() {
        return Car.class;
    }
    
    @Override
    public boolean isSingleton() {
        return true;
    }
    
    // setter for carInfo
    public void setCarInfo(String carInfo) {
        this.carInfo = carInfo;
    }
}

public class MyFactoryBean implements FactoryBean<MyBean> {
    private String property1;
    private int property2;
    
    // 设置属性
    public void setProperty1(String property1) {
        this.property1 = property1;
    }
    
    public void setProperty2(int property2) {
        this.property2 = property2;
    }
    
    @Override
    public MyBean getObject() throws Exception {
        // 创建并返回实际的 bean 实例
        MyBean myBean = new MyBean();
        myBean.setProp1(property1);
        myBean.setProp2(property2);
        return myBean;
    }
    
    @Override
    public Class<?> getObjectType() {
        return MyBean.class;
    }
    
    @Override
    public boolean isSingleton() {
        return true; // 通常 FactoryBean 返回单例
    }
}

@Configuration
public class AppConfig {
    @Bean
    public MyFactoryBean myBean() {
        MyFactoryBean factory = new MyFactoryBean();
        factory.setProperty1("someValue");
        factory.setProperty2(123);
        return factory;
    }
}

// 获取 FactoryBean 创建的对象
MyBean myBean = context.getBean("myBean", MyBean.class);

// 如果要获取 FactoryBean 本身而不是它创建的对象，需要在 bean id 前加 &
FactoryBean<?> factoryBean = context.getBean("&myBean", FactoryBean.class);

结合其他Spring特性

public class TimerFactoryBean implements FactoryBean<Timer>, InitializingBean, DisposableBean {
    private Timer timer;
    private String timerName;
    
    @Override
    public void afterPropertiesSet() throws Exception {
        this.timer = new Timer(timerName);
    }
    
    @Override
    public Timer getObject() throws Exception {
        return this.timer;
    }
    
    @Override
    public void destroy() throws Exception {
        this.timer.cancel();
    }
    
    // getter/setter...
}
